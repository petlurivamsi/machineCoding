Table follows {
  following_user_id integer
  followed_user_id integer
  created_at timestamp
}

Table users {
  id integer [primary key]
  firstName varchar
  lastName varchar
  username varchar
  email varchar
  role varchar
  hashed_password varchar
  school varchar
  mobile varchar
  dob date
  isOnline boolean
  user_profile_pic varchar
  created_at timestamp
}


Table blockedUsers {
  user_id integer [primary key]
  blocked_user_id integer [primary key]
  blocked_at timestamp
}

Table likes {
  liked_by_id integer [primary key]
  liked_to_id integer [primary key]
  liked_at timestamp
}

Table posts {
  id integer [primary key]
  title varchar
  body text [note: 'Content of the post']
  user_id integer
  status varchar
  image_details varchar
  created_at timestamp
}

Table comments {
  id integer [primary key]
  description varchar
  user_id integer
  parent_comment_id integer
  post_id integer
  created_at timestamp
}

Ref: follows.following_user_id <> users.id

Ref: follows.followed_user_id <> users.id

Ref: likes.liked_by_id <> users.id

Ref: likes.liked_to_id <> posts.id

Ref: blockedUsers.blocked_user_id <> users.id

Ref: blockedUsers.user_id <> users.id

Ref: comments.user_id > users.id

Ref: comments.post_id > posts.id

Ref: comments.parent_comment_id > comments.id
